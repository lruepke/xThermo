# Developer test programs (XTHERMO_BUILD_TESTS=ON)
foreach(TEST_ITEM test_gsl test_PROST)
  add_executable(${TEST_ITEM} ${TEST_ITEM}.cpp)
  target_link_libraries(${TEST_ITEM} PRIVATE xThermal)
  add_test(NAME ${TEST_ITEM} COMMAND ${TEST_ITEM})
endforeach()
target_include_directories(test_PROST PRIVATE ${PROJECT_SOURCE_DIR}/ThirdParties/PROST)
if(XTHERMO_OPENMP)
  add_executable(test_omp test_omp.cpp)
  target_link_libraries(test_omp PRIVATE xThermal)
  add_test(NAME test_omp COMMAND test_omp)
endif()
# interactive per-module programs (not registered as tests)
foreach(Module H2O NaCl H2ONaCl)
  add_executable(test_${Module} ${PROJECT_SOURCE_DIR}/Library/${Module}/test.cxx)
  target_link_libraries(test_${Module} PRIVATE xThermal)
  target_compile_options(test_${Module} PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang>:-Wno-writable-strings>)
endforeach()
